# Generations Covered Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Next.js app that computes a patient's hospital financial-assistance eligibility from published policy data and produces a mailable application PDF, in a 2-hour hackathon build.

**Architecture:** One Next.js app. A pure TypeScript engine computes eligibility from JSON data files; three route handlers wrap it; two client pages consume the routes. A frozen `types.ts` plus committed JSON fixtures are the contract between the API track (Tasks 1 to 9, Gideon) and the UI track (Tasks 10 to 15, Avery). The UI builds against fixtures until the 1:05 integration checkpoint.

**Tech Stack:** Next.js 15 (App Router, TypeScript, Tailwind, `src/` dir), pdf-lib, Vitest, tsx. Deployed on Vercel from `main`.

**Spec:** `docs/superpowers/specs/2026-10-10-covered-design.md`

## Global Constraints

- Node 20 or newer. npm.
- No language model anywhere in the product. Eligibility comes only from `data/hospitals.json` and `data/fpl-2026.json`.
- `src/lib/types.ts` is frozen after Task 1 is pushed. Any later change is announced in chat before push.
- Display name constants: `APP_NAME = "Generations Covered"`, `APP_SHORT = "Covered"` in `src/lib/config.ts`.
- Exactly two routes: `/` and `/result`. No database, no accounts, no state library.
- Result query params are exactly: `hospital`, `size`, `income`, `insurance`, `county`, `bill`. State is always `"GA"`.
- Dependencies beyond the scaffold: `pdf-lib` (runtime), `vitest` and `tsx` (dev). Nothing else without a chat message.
- Every policy figure on screen links to the hospital's `source_url`.
- `discount_pct: null` means "eligible, amount set by hospital" and must never be rendered as a dollar figure.
- Ownership: Gideon edits `src/lib/**`, `src/app/api/**`, `data/**`, `tests/**`, `scripts/**`, `package.json`. Avery edits `src/app/page.tsx`, `src/app/result/**`, `src/components/**`, `src/app/layout.tsx`, `src/app/globals.css`.
- Hard freeze at 1:50. Cut list order: real form fill, generated PDF, result-page source links, third hospital.

## Review Focus

1. Non-integer household size (`3.5`) must be rejected with a 400, not silently rounded or looked up as `by_size["3.5"]` and crash. Pinned in Task 4.
2. County entered with odd casing or whitespace (`" fulton "`) must still pass Grady's residency check. Pinned in Task 4.
3. A bill with cents (`9400.50`) at a 35% tier must round to cents (`6110.33`), not produce a floating-point tail. Pinned in Task 4.
4. A `/result` URL with a missing or garbage number (`income=abc`) must produce a 400 with a readable error, not a 500 or `NaN%` on screen. Pinned in Task 6 (API) and shown in Task 13 (UI error state).
5. An application request with an empty applicant field must 400 before any PDF is rendered. Pinned in Task 8.

```ts
import type { FplTable, HospitalPolicy } from "./types";

const APPLIES = new Set(["uninsured", "underinsured", "any"]);

export function validateHospitals(hospitals: HospitalPolicy[]): string[] {
  const errors: string[] = [];
  const seen = new Set<string>();
  hospitals.forEach((h, i) => {
    const at = `hospitals[${i}] (${h.id})`;
    if (!h.id || h.id !== h.id.toLowerCase()) errors.push(`${at}: id must be lowercase`);
    if (seen.has(h.id)) errors.push(`${at}: duplicate id`);
    seen.add(h.id);
    for (const [k, url] of Object.entries({ policy_url: h.policy_url, application_url: h.application_url })) {
      if (!url.startsWith("https://")) errors.push(`${at}: ${k} must start with https://`);
    }
    for (const c of h.policy_conflicts) {
      if (!c.source_url.startsWith("https://")) errors.push(`${at}: policy_conflicts source_url must start with https://`);
    }
    let lastMax = -Infinity;
    let seenNull = false;
    h.tiers.forEach((t, j) => {
      const tat = `${at} tiers[${j}]`;
      if (!APPLIES.has(t.applies_to)) errors.push(`${tat}: applies_to invalid`);
      if (t.discount_pct !== null && (t.discount_pct < 0 || t.discount_pct > 100)) errors.push(`${tat}: discount_pct must be 0..100 or null`);
      if (t.max_fpl_pct === null) {
        seenNull = true;
      } else {
        if (seenNull) errors.push(`${tat}: tiers must be sorted ascending with null last`);
        if (t.max_fpl_pct < lastMax) errors.push(`${tat}: tiers must be sorted ascending by max_fpl_pct`);
        lastMax = t.max_fpl_pct;
      }
    });
  });
  return errors;
}

export function validateFpl(table: FplTable): string[] {
  const errors: string[] = [];
  let last = 0;
  for (let s = 1; s <= 8; s++) {
    const v = table.by_size[String(s)];
    if (typeof v !== "number") {
      errors.push(`by_size.${s} missing`);
      continue;
    }
    if (v <= last) errors.push(`by_size.${s} must be increasing`);
    last = v;
  }
  if (!(table.additional_person > 0)) errors.push("additional_person must be > 0");
  if (!table.source_url.startsWith("https://")) errors.push("source_url must start with https://");
  return errors;
}
```

```ts
import type {
  EligibilityInput, EligibilityResult, EligibilityStatus, FplTable,
  HospitalPolicy, InsuranceStatus, Residency, Tier,
} from "./types";

export class ValidationError extends Error {}

const INSURANCE: ReadonlySet<string> = new Set(["uninsured", "underinsured"]);

export function validateInput(input: EligibilityInput): void {
  if (!Number.isInteger(input.household_size) || input.household_size < 1 || input.household_size > 20) {
    throw new ValidationError("household_size must be a whole number from 1 to 20");
  }
  if (!Number.isFinite(input.annual_income) || input.annual_income < 0) {
    throw new ValidationError("annual_income must be a number of 0 or more");
  }
  if (!Number.isFinite(input.bill_amount) || input.bill_amount <= 0) {
    throw new ValidationError("bill_amount must be a number greater than 0");
  }
  if (!INSURANCE.has(input.insurance_status)) {
    throw new ValidationError('insurance_status must be "uninsured" or "underinsured"');
  }
  if (typeof input.county !== "string" || input.county.trim() === "") {
    throw new ValidationError("county is required");
  }
}

export function fplBase(householdSize: number, table: FplTable): number {
  if (householdSize <= 8) return table.by_size[String(householdSize)];
  return table.by_size["8"] + (householdSize - 8) * table.additional_person;
}

export function matchTier(tiers: Tier[], fplPct: number, status: InsuranceStatus): Tier | null {
  for (const t of tiers) {
    const statusOk = t.applies_to === "any" || t.applies_to === status;
    const pctOk = t.max_fpl_pct === null || fplPct <= t.max_fpl_pct;
    if (statusOk && pctOk) return t;
  }
  return null;
}

const norm = (s: string) => s.trim().toLowerCase();

export function residencyOk(residency: Residency, input: EligibilityInput): boolean {
  switch (residency.type) {
    case "none":
      return true;
    case "state":
      return residency.values.map(norm).includes(norm(input.state));
    case "county":
      return residency.values.map(norm).includes(norm(input.county));
  }
}

const round2 = (n: number) => Math.round(n * 100) / 100;

export function computeEligibility(
  input: EligibilityInput,
  hospital: HospitalPolicy,
  table: FplTable,
): EligibilityResult {
  validateInput(input);
  const fpl_base = fplBase(input.household_size, table);
  const fpl_pct = Math.round((input.annual_income / fpl_base) * 100);
  const tier = matchTier(hospital.tiers, fpl_pct, input.insurance_status);

  let status: EligibilityStatus;
  let bill_after: number | null;
  if (!tier) {
    status = "no_match";
    bill_after = null;
  } else if (tier.discount_pct === null) {
    status = "eligible_sliding_scale";
    bill_after = null;
  } else if (tier.discount_pct === 100) {
    status = "free";
    bill_after = 0;
  } else {
    status = "discounted";
    bill_after = round2(input.bill_amount * (1 - tier.discount_pct / 100));
  }

  return {
    hospital: hospital.id,
    hospital_name: hospital.name,
    household_size: input.household_size,
    annual_income: input.annual_income,
    insurance_status: input.insurance_status,
    county: input.county,
    state: input.state,
    fpl_base,
    fpl_pct,
    tier_matched: tier,
    status,
    bill_before: input.bill_amount,
    bill_after,
    residency_ok: residencyOk(hospital.residency, input),
    residency: hospital.residency,
    documents_required: hospital.documents_required,
    application_url: hospital.application_url,
    mailing_address: hospital.mailing_address,
    source_url: hospital.policy_url,
    policy_date: hospital.policy_date,
    policy_conflicts: hospital.policy_conflicts,
    verified: hospital.verified,
  };
}
```

```ts
import { ValidationError } from "./engine";
import type { Applicant, EligibilityInput, InsuranceStatus } from "./types";

type Obj = Record<string, unknown>;
const obj = (x: unknown): Obj => (x && typeof x === "object" ? (x as Obj) : {});
const num = (x: unknown): number => (x === "" || x === null || x === undefined ? NaN : Number(x));
const str = (x: unknown): string => (x === null || x === undefined ? "" : String(x));

export function parseInput(body: unknown): EligibilityInput {
  const b = obj(body);
  return {
    hospital_id: str(b.hospital_id),
    household_size: num(b.household_size),
    annual_income: num(b.annual_income),
    bill_amount: num(b.bill_amount),
    insurance_status: str(b.insurance_status) as InsuranceStatus,
    county: str(b.county),
    state: str(b.state) || "GA",
  };
}

export function parseApplicant(body: unknown): Applicant {
  const b = obj(body);
  const a: Applicant = {
    full_name: str(b.full_name).trim(),
    mailing_address: str(b.mailing_address).trim(),
    phone: str(b.phone).trim(),
    date_of_birth: str(b.date_of_birth).trim(),
  };
  for (const [k, v] of Object.entries(a)) {
    if (v === "") throw new ValidationError(`applicant.${k} is required`);
  }
  return a;
}
```

```ts
import { computeEligibility, ValidationError } from "@/lib/engine";
import { parseInput } from "@/lib/parse";

export async function POST(req: Request) {
  let body: unknown;
  try {
    body = await req.json();
  } catch {
    return NextResponse.json({ error: "Body must be JSON" }, { status: 400 });
  }
  const input = parseInput(body);
  const hospital = getHospital(input.hospital_id);
  if (!hospital) {

  }
  try {
    return NextResponse.json(computeEligibility(input, hospital, getFplTable()));

    throw e;
```

```ts
import { PDFDocument, PDFFont, StandardFonts, rgb } from "pdf-lib";
import { APP_NAME } from "../config";
import type { Applicant, EligibilityResult } from "../types";

const usd = (n: number) =>
  n.toLocaleString("en-US", { style: "currency", currency: "USD", maximumFractionDigits: 0 });
const clip = (s: string) => (s.length > 90 ? s.slice(0, 87) + "..." : s);

const INSURANCE_LABEL = {
  uninsured: "No insurance",
  underinsured: "Insured, unable to afford the bill",
} as const;

export function tierDescription(r: EligibilityResult): string {
  const t = r.tier_matched;
  if (!t) return "No published tier matched this household.";
  const cap = t.max_fpl_pct === null ? "any income above the free-care line" : `up to ${t.max_fpl_pct}% of the Federal Poverty Level`;
  const amount = t.discount_pct === null ? "discount set by the hospital's sliding scale" : `${t.discount_pct}% discount`;
  return `${amount}, ${cap}${t.note ? `. ${t.note}` : ""}`;
}

export async function renderApplicationPdf(r: EligibilityResult, a: Applicant): Promise<Uint8Array> {
  const doc = await PDFDocument.create();
  const page = doc.addPage([612, 792]);
  const font = await doc.embedFont(StandardFonts.Helvetica);
  const bold = await doc.embedFont(StandardFonts.HelveticaBold);
  const left = 54;
  let y = 740;

  const line = (text: string, opts: { size?: number; font?: PDFFont; gap?: number; x?: number } = {}) => {
    const size = opts.size ?? 11;
    page.drawText(clip(text), { x: opts.x ?? left, y, size, font: opts.font ?? font });
    y -= size + (opts.gap ?? 5);
  };
  const heading = (text: string) => {
    y -= 8;
    line(text, { size: 13, font: bold });
  };

  line(`Financial Assistance Application: ${r.hospital_name}`, { size: 18, font: bold, gap: 4 });
  line(`Prepared with ${APP_NAME} on ${new Date().toISOString().slice(0, 10)}`, { size: 9, gap: 10 });

  heading("Applicant");
  line(`Name: ${a.full_name}`);
  a.mailing_address.split("\n").forEach((addr, i) => line(i === 0 ? `Mailing address: ${addr}` : addr, { x: i === 0 ? left : left + 92 }));
  line(`Phone: ${a.phone}`);
  line(`Date of birth: ${a.date_of_birth}`);

  heading("Household");
  line(`Household size: ${r.household_size}`);
  line(`Annual household income: ${usd(r.annual_income)}`);
  line(`Insurance: ${INSURANCE_LABEL[r.insurance_status]}`);
  line(`County of residence: ${r.county}, ${r.state}`);

  heading("Eligibility under the hospital's published policy");
  line(`Income is ${r.fpl_pct}% of the Federal Poverty Level (FPL for ${r.household_size}: ${usd(r.fpl_base)})`);
  line(`Tier: ${tierDescription(r)}`);
  line(`Bill before assistance: ${usd(r.bill_before)}`);
  line(`Estimated bill after assistance: ${r.bill_after === null ? "Determined by the hospital" : usd(r.bill_after)}`);

  heading("Documents to include");
  for (const d of r.documents_required) {
    page.drawRectangle({ x: left, y: y - 1, width: 10, height: 10, borderWidth: 1, borderColor: rgb(0, 0, 0) });
```

```ts
import { existsSync, readFileSync } from "node:fs";
import path from "node:path";
import { PDFDocument } from "pdf-lib";
import type { Applicant, EligibilityResult } from "../types";

export function applicationValues(r: EligibilityResult, a: Applicant): Record<string, string> {
  return {
    full_name: a.full_name,
    mailing_address: a.mailing_address,
    phone: a.phone,
    date_of_birth: a.date_of_birth,
    household_size: String(r.household_size),
    annual_income: String(r.annual_income),
    insurance_status: r.insurance_status === "uninsured" ? "No insurance" : "Insured",
    county: r.county,
    bill_amount: String(r.bill_before),
  };
}

export async function fillHospitalForm(r: EligibilityResult, a: Applicant): Promise<Uint8Array | null> {
  const dir = path.join(process.cwd(), "data", "forms");
  const mapPath = path.join(dir, `${r.hospital}-fields.json`);
  const pdfPath = path.join(dir, `${r.hospital}.pdf`);
  if (!existsSync(mapPath) || !existsSync(pdfPath)) return null;

  const map: Record<string, string> = JSON.parse(readFileSync(mapPath, "utf8"));
  const values = applicationValues(r, a);
  const doc = await PDFDocument.load(readFileSync(pdfPath));
  const form = doc.getForm();
  for (const [ours, theirs] of Object.entries(map)) {
    const v = values[ours];
    if (v !== undefined) form.getTextField(theirs).setText(v);
  }
  form.flatten();
  return doc.save();
}
```

```tsx
"use client";

import { Suspense, useEffect, useState } from "react";
import { useSearchParams } from "next/navigation";
import type { EligibilityInput, EligibilityResult, HospitalPolicy, InsuranceStatus } from "@/lib/types";
import { ResultView } from "@/components/ResultView";

function inputFromParams(sp: URLSearchParams): EligibilityInput {
  return {
    hospital_id: sp.get("hospital") ?? "",
    household_size: Number(sp.get("size")),
    annual_income: Number(sp.get("income")),
    bill_amount: Number(sp.get("bill")),
    insurance_status: (sp.get("insurance") ?? "uninsured") as InsuranceStatus,
    county: sp.get("county") ?? "",
    state: "GA",
  };
}

function ResultLoader() {
  const sp = useSearchParams();
  const input = inputFromParams(sp);
  const [result, setResult] = useState<EligibilityResult | null>(null);
  const [hospitals, setHospitals] = useState<HospitalPolicy[]>([]);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    fetch("/api/hospitals").then((r) => r.json()).then((d) => setHospitals(d.hospitals));
  }, []);

  useEffect(() => {
    setResult(null);
    setError(null);
    fetch("/api/eligibility", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(input),
    })
      .then(async (r) => {
        if (!r.ok) throw new Error((await r.json()).error ?? `Request failed (${r.status})`);
        return r.json();
      })
      .then(setResult)
      .catch((e: Error) => setError(e.message));
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [sp.toString()]);

  if (error) {
    return (
      <div>
        <p className="text-red-800">Something is off with the numbers: {error}</p>
        <a className="mt-2 inline-block" href="/">Go back and try again</a>
      </div>
    );
  }
  if (!result) return <p className="text-stone-600">Checking {input.hospital_id}'s policy...</p>;
  return <ResultView r={result} hospitals={hospitals} />;
}

export default function ResultPage() {
  return (
    <Suspense fallback={null}>
      <ResultLoader />
    </Suspense>
  );
}
```

```tsx
"use client";

import { useState } from "react";
import type { Applicant, EligibilityInput, EligibilityResult } from "@/lib/types";

const field = "mt-1 block w-full rounded border border-stone-300 bg-white px-3 py-2";

export function ApplicationPanel({ r, input, pdfEnabled = true }: { r: EligibilityResult; input: EligibilityInput; pdfEnabled?: boolean }) {
  const [open, setOpen] = useState(false);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const [a, setA] = useState<Applicant>({ full_name: "", mailing_address: "", phone: "", date_of_birth: "" });
  const set = (k: keyof Applicant) => (e: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => setA({ ...a, [k]: e.target.value });

  async function download() {
    setBusy(true);
    setError(null);
    try {
      const res = await fetch("/api/application", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ input, applicant: a }),
      });
      if (!res.ok) throw new Error((await res.json()).error ?? `Request failed (${res.status})`);
      const blob = await res.blob();
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = `covered-${input.hospital_id}-application.pdf`;
      link.click();
      URL.revokeObjectURL(url);
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <section className="space-y-4">
      <h2 className="text-lg font-semibold">Apply</h2>
      {pdfEnabled && !open && (
        <button className="rounded px-4 py-3 font-semibold text-white" style={{ background: "var(--accent)" }} onClick={() => setOpen(true)}>
          Prepare my application
        </button>
      )}
      {pdfEnabled && open && (
        <div className="space-y-3">
          <label className="block"><span className="font-medium">Full name</span><input className={field} value={a.full_name} onChange={set("full_name")} /></label>
          <label className="block"><span className="font-medium">Mailing address</span><textarea className={field} rows={3} value={a.mailing_address} onChange={set("mailing_address")} /></label>
          <label className="block"><span className="font-medium">Phone</span><input className={field} value={a.phone} onChange={set("phone")} /></label>
          <label className="block"><span className="font-medium">Date of birth</span><input className={field} type="date" value={a.date_of_birth} onChange={set("date_of_birth")} /></label>
          <button disabled={busy} className="rounded px-4 py-3 font-semibold text-white disabled:opacity-50" style={{ background: "var(--accent)" }} onClick={download}>
            {busy ? "Preparing..." : "Download application PDF"}
          </button>
          {error && <p className="text-red-800">{error}</p>}
        </div>
      )}
      <div>
        <p className="font-medium">Documents to include</p>
        <ul className="mt-1 space-y-1">
          {r.documents_required.map((d) => (
            <li key={d} className="flex items-start gap-2"><span className="mt-1 inline-block h-4 w-4 shrink-0 border border-stone-500" />{d}</li>
          ))}
        </ul>
        <a className="mt-3 inline-block text-sm" href={r.application_url} target="_blank" rel="noreferrer">Open {r.hospital_name}'s official application</a>
      </div>
    </section>
  );
}
```
